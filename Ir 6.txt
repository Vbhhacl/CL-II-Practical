import numpy as np
import networkx as nx
def pagerank_random_walk(adj_matrix, damping=0.85, max_iter=100, tol=1e-6):
    n = adj_matrix.shape[0]
    out_degree = adj_matrix.sum(axis=0)
    dangling = (out_degree == 0)
    for i in range(n):
        if dangling[i]:
            adj_matrix[:, i] = 1.0 / n
    out_degree = adj_matrix.sum(axis=0)
    M = adj_matrix / out_degree
    ranks = np.ones(n) / n
    for _ in range(max_iter):
        new_ranks = (1 - damping) / n + damping * M @ ranks
        if np.linalg.norm(new_ranks - ranks, 1) < tol:
            break
        ranks = new_ranks
    return ranks
def get_user_graph():
    n = int(input("Enter the number of pages (nodes): "))
    print("Pages are numbered from 0 to", n-1)
    print("Enter directed edges in format: source target")
    print("Means: source page links to target page.")
    print("Type 'done' when finished entering edges.")
    edges = []
    while True:
        inp = input("Edge: ").strip()
        if inp.lower() == 'done':
            break
        try:
            src, dst = map(int, inp.split())
            if 0 <= src < n and 0 <= dst < n:
                edges.append((src, dst))
            else:
                print(f"Invalid nodes. Must be between 0 and {n-1}. Try again.")
        except:
            print("Invalid input format. Please enter two integers separated by space.")
    return n, edges
def build_adjacency(n, edges):
    adj = np.zeros((n, n))
    # adj[i][j] = 1 if j -> i (j links to i)
    for src, dst in edges:
        adj[dst][src] = 1
    return adj
if __name__ == "__main__":
    n, edges = get_user_graph()
    adj = build_adjacency(n, edges)
    # Run custom PageRank
    pr_rw = pagerank_random_walk(adj)
    sorted_nodes_rw = np.argsort(-pr_rw)
    print("\nPageRank using Random Walk Method:")
    print(sorted_nodes_rw)
    # Run NetworkX PageRank
    G = nx.DiGraph()
    G.add_edges_from(edges)
    pr_nx = nx.pagerank(G, alpha=0.85)
    sorted_nodes_nx = sorted(pr_nx, key=pr_nx.get, reverse=True)
    print("\nPageRank using inbuilt pagerank method:")
    print(", ".join(str(n) for n in sorted_nodes_nx) + ",")
