import re
from collections import defaultdict
def preprocess(text):
    text = text.lower()
    words = re.findall(r'\b\w+\b', text)
    return words
def build_inverted_index(docs):
    inverted_index = defaultdict(set)
    for doc_id, text in docs.items():
        words = preprocess(text)
        for word in words:
            inverted_index[word].add(doc_id)
    return inverted_index

def search(query, inverted_index):
    query_terms = preprocess(query)
    result_sets = [inverted_index.get(term, set()) for term in query_terms]
    if result_sets:
        result_docs = set.intersection(*result_sets)
    else:
        result_docs = set()
    return result_docs

if __name__ == "__main__":
    documents = {
        1: "The quick brown fox jumps over the lazy dog",
        2: "Never jump over the lazy dog quickly",
        3: "A quick brown dog outpaces a quick fox"
    }
    inverted_index = build_inverted_index(documents)
    print("Inverted Index:")
    for term, doc_ids in inverted_index.items():
        print(f"{term}: {sorted(doc_ids)}")
    while True:
        print("\nMenu:")
        print("1. Search documents")
        print("2. Exit")
        choice = input("Enter choice (1 or 2): ").strip()
        if choice == '1':
            query = input("Enter search query: ").strip()
            results = search(query, inverted_index)
            if results:
                print(f"Documents containing '{query}': {sorted(results)}")
            else:
                print(f"No documents found containing '{query}'.")
        elif choice == '2':
            print("Exiting the program. Goodbye!")
            break
        else:
            print("Invalid choice. Please enter 1 or 2.")
